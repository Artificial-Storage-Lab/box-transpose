Key Insight
Global memory is the slowest! Use shared memory to cache data and reduce global accesses.
How It All Works Together
  1. CPU allocates data in Host RAM and copies to GPU Global Memory (cudaMemcpy)
  2. CPU launches kernel: kernel<<<blocks, threads>>>(args)
  3. GPU scheduler distributes blocks across available SMs
  4. Each SM organizes its threads into warps (32 threads each)
  5. Warps execute in lockstep (SIMD) - same instruction, different data
  6. Threads access memory hierarchy: Registers → Shared → L1 → L2 → Global
  7. Results written back to Global Memory
  8. CPU copies results back to Host RAM (cudaMemcpy)

Launch Configuration & Thread ID Calculation
// Define configuration
int threadsPerBlock = 256;
int blocksPerGrid = (N + threadsPerBlock - 1) / threadsPerBlock;

// Launch kernel
vectorAdd<<>>(d_a, d_b, d_c, N);

// Inside kernel - calculate unique thread ID
__global__ void vectorAdd(float* a, float* b, float* c, int n) {
    int tid = threadIdx.x + blockIdx.x * blockDim.x;
    
    // Guard clause - critical!
    if (tid < n) {
        c[tid] = a[tid] + b[tid];
    }
}
Blocks vs Warps - Critical Distinction
Blocks: Defined by programmer (up to 1024 threads), can synchronize and share memory
Warps: Fixed hardware groups of exactly 32 threads, execute in lockstep automatically

Coalesced Access

Adjacent threads accessing adjacent memory addresses in a single transaction. 

// GOOD - Coalesced
int tid = threadIdx.x + blockIdx.x * blockDim.x;
float val = data[tid];  // Sequential

// BAD - Non-coalesced
float val = data[tid * stride];  // Scattered
Shared Memory Usage

Cache frequently accessed data to avoid slow global memory. 

__shared__ float tile[256];

// Load from global
tile[threadIdx.x] = global[idx];
__syncthreads();

// Process from shared (fast!)
result = compute(tile[threadIdx.x]);
Register Spilling

Too many local variables overflow from fast registers to slow local memory.

Cause Too many variables
Effect Local memory use
Penalty 400-800 cycle latency
Solution Reduce variables

Memory Transfer Bottleneck
Host ↔ Device transfers via PCIe are often the biggest bottleneck. A computation that takes 1ms on GPU might need 100ms of transfer time! Strategies:
  • Minimize transfers - compute multiple steps on GPU
  • Use asynchronous transfers with CUDA streams
  • Overlap transfer with computation
  • Use pinned (page-locked) memory for faster transfers

Complete GPU Program Execution Flow

1
Initialize & Allocate Host Memory
Allocate arrays in system RAM, initialize data 
float *h_a = (float*)malloc(size);
float *h_b = (float*)malloc(size);
float *h_c = (float*)malloc(size);
2
Allocate Device Memory
Allocate corresponding arrays in GPU global memory 
float *d_a, *d_b, *d_c;
cudaMalloc((void**)&d_a, size);
cudaMalloc((void**)&d_b, size);
cudaMalloc((void**)&d_c, size);
3
Copy Data Host → Device
Transfer input data from system RAM to GPU via PCIe (slow!) 
cudaMemcpy(d_a, h_a, size, cudaMemcpyHostToDevice);
cudaMemcpy(d_b, h_b, size, cudaMemcpyHostToDevice);
4
Launch Kernel
Configure and launch parallel execution on GPU 
int threads = 256;
int blocks = (N + threads - 1) / threads;
vectorAdd<<>>(d_a, d_b, d_c, N);
5
GPU Execution
Scheduler distributes blocks to SMs → SMs organize into warps → Warps execute in lockstep
Inside each thread:
1. Calculate unique thread ID
2. Check bounds (guard clause)
3. Access memory (Registers → Shared → L1 → L2 → Global)
4. Perform computation
5. Write result to global memory
6
Synchronize
Wait for GPU to finish (kernel launch is asynchronous!) 
cudaDeviceSynchronize();  // Block until GPU done
7
Copy Results Device → Host
Transfer results back from GPU to system RAM via PCIe 
cudaMemcpy(h_c, d_c, size, cudaMemcpyDeviceToHost);
8
Cleanup
Free device and host memory to prevent leaks 
cudaFree(d_a);
cudaFree(d_b);
cudaFree(d_c);
free(h_a);
free(h_b);
free(h_c);
Complete Example: Vector Addition
// Kernel definition
__global__ void vectorAdd(float* a, float* b, float* c, int n) {
    int tid = threadIdx.x + blockIdx.x * blockDim.x;
    if (tid < n) {
        c[tid] = a[tid] + b[tid];
    }
}

// Host code
int main() {
    int N = 10000;
    size_t size = N * sizeof(float);
    
    // 1. Allocate host memory
    float *h_a = (float*)malloc(size);
    float *h_b = (float*)malloc(size);
    float *h_c = (float*)malloc(size);
    
    // Initialize h_a and h_b...
    
    // 2. Allocate device memory
    float *d_a, *d_b, *d_c;
    cudaMalloc(&d_a, size);
    cudaMalloc(&d_b, size);
    cudaMalloc(&d_c, size);
    
    // 3. Copy to device
    cudaMemcpy(d_a, h_a, size, cudaMemcpyHostToDevice);
    cudaMemcpy(d_b, h_b, size, cudaMemcpyHostToDevice);
    
    // 4. Launch kernel
    int threads = 256;
    int blocks = (N + threads - 1) / threads;
    vectorAdd<<>>(d_a, d_b, d_c, N);
    
    // 6. Synchronize
    cudaDeviceSynchronize();
    
    // 7. Copy results back
    cudaMemcpy(h_c, d_c, size, cudaMemcpyDeviceToHost);
    
    // 8. Cleanup
    cudaFree(d_a); cudaFree(d_b); cudaFree(d_c);
    free(h_a); free(h_b); free(h_c);
    
    return 0;
}
